}

\codelabel

\begin{code}
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[] = "Hello, world!";
    char *start = str, *end = str + strlen(str) - 1;
    while (start < end) {
        char temp = *start;
        *start++ = *end;
        *end-- = temp;
    }
    cout << str;
    return 0;
}

\program{

}

\codelabel

\begin{code}
#include <iostream>
using namespace std;

int main() {
    string sentence = "This is a sentence with a long word.", word, longest;
    for (char c : sentence + " ") {
        if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'))
            word += c;
        else {
            if (word.length() > longest.length())
                longest = word;
            word = "";
        }
    }
    cout << "Longest word is: " << longest;
    return 0;
}

\program{

}

\codelabel

\begin{code}
#include <iostream>
#include <fstream>
#include <vector>
using namespace std;

int main() {
    ofstream file("file.txt");
    vector<string> lines;
    string text;
    while (getline(cin, text) && text != "EXIT") {
        file << text << endl;
        lines.push_back(text);
    }
    for (int i = lines.size() - 1; i >= 0; i--)
        cout << lines[i] << endl;
    return 0;
}

\program{
    8. Write a program to find the union and intersection of arrays.
}

\codelabel

\begin{code}
#include <iostream>
using namespace std;

int main() {
    int n1, n2, a[20], b[20];
    cin >> n1;
    for (int i = 0; i < n1; i++) cin >> a[i];
    cin >> n2;
    for (int i = 0; i < n2; i++) cin >> b[i];

    cout << "Intersection: ";
    for (int i = 0; i < n1; i++)
        for (int j = 0; j < n2; j++)
            if (a[i] == b[j]) { cout << a[i] << " "; break; }

    cout << endl << "Union: ";
    for (int i = 0; i < n1; i++) cout << a[i] << " ";
    for (int j = 0; j < n2; j++) {
        bool found = false;
        for (int i = 0; i < n1; i++)
            if (b[j] == a[i]) found = true;
        if (!found) cout << b[j] << " ";
    }
    return 0;
}
\end{code}

\program{
    9. Write a program to count frequency of each character (case insensitive).
}

\codelabel

\begin{code}
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    ifstream file("file.txt");
    char ch;
    int freq[26] = {0};
    while (file.get(ch)) {
        if (ch >= 'A' && ch <= 'Z') ch += 32;
        if (ch >= 'a' && ch <= 'z') freq[ch - 'a']++;
    }
    for (int i = 0; i < 26; i++)
        if (freq[i] > 0)
            cout << char(i + 'a') << " : " << freq[i] << endl;
    return 0;
}
\end{code}
